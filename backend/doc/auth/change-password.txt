================================================================
  CHANGE PASSWORD API — POST /v1/auth/change-password
  Base URL: http://localhost:3001
================================================================


--------------------------------------------------
1. NO ACCESS TOKEN (401)
--------------------------------------------------
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -d '{ "oldPassword": "Secret123", "newPassword": "NewSecret456" }'

Expected: 401
Response: { "success": false, "message": "Unauthorized" }


--------------------------------------------------
2. VALIDATION ERRORS (Zod)
--------------------------------------------------

--- Missing oldPassword ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "newPassword": "NewSecret456" }'

Expected: 422


--- Missing newPassword ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123" }'

Expected: 422


--- newPassword too short ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123", "newPassword": "Ab1" }'

Expected: 422


--- newPassword missing uppercase ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123", "newPassword": "newsecret123" }'

Expected: 422


--- newPassword missing lowercase ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123", "newPassword": "NEWSECRET123" }'

Expected: 422


--- newPassword missing number ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123", "newPassword": "NewSecretPass" }'

Expected: 422


--- Empty body ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{}'

Expected: 422


--- newPassword fails all conditions (no uppercase, no number) ---
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123", "newPassword": "abcdefgh" }'

Expected: 422


--------------------------------------------------
3. WRONG CURRENT PASSWORD (401)
--------------------------------------------------
curl -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "WrongPass1", "newPassword": "NewSecret456" }'

Expected: 401
Response: { "success": false, "message": "Current password is incorrect" }


--------------------------------------------------
4. RATE LIMIT (429)
--------------------------------------------------
for i in {1..6}; do
  curl -s -o /dev/null -w "%{http_code}\n" \
    -X POST http://localhost:3001/v1/auth/change-password \
    -H "Content-Type: application/json" \
    -b cookies.txt \
    -d '{ "oldPassword": "Secret123", "newPassword": "NewSecret456" }'
done

Expected: first 5 return 200, 6th returns 429
Response: { "success": false, "message": "Too many requests, please try again later" }


--------------------------------------------------
5. SUCCESSFUL CHANGE PASSWORD (200)
--------------------------------------------------
curl -v -X POST http://localhost:3001/v1/auth/change-password \
  -H "Content-Type: application/json" \
  -b cookies.txt \
  -d '{ "oldPassword": "Secret123", "newPassword": "NewSecret456" }'

Expected: 200
Response: { "success": true, "message": "Password changed successfully", "data": null }

Cookies cleared:
  Set-Cookie: accessToken=; Path=/; Max-Age=0
  Set-Cookie: refreshToken=; Path=/v1/auth; Max-Age=0

DB: password updated, all refresh_tokens revoked


--------------------------------------------------
6. OLD TOKENS REVOKED AFTER CHANGE
--------------------------------------------------
-- After changing password, try old refresh token:
curl -X POST http://localhost:3001/v1/auth/refresh-token \
  -b cookies-with-old-refresh.txt

Expected: 401
Response: { "success": false, "message": "Refresh token revoked" }
